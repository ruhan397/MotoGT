const Appointment = require('../models/Appointment');

// Friday is the weekly workshop holiday
const isFriday = (dateString) => {
  const date = new Date(`${dateString}T12:00:00`);
  return date.getDay() === 5;
};

// Get or create workshop settings
const getWorkshopSettings = async () => {
  let settings = await WorkshopSettings.findOne();

  if (!settings) {
    settings = await WorkshopSettings.create({
      maxVehicles: 5,
      dailyWorkCapacity: 2,
      approvalLock: 0
    });
  }

  return settings;
};

// Format date as YYYY-MM-DD
const formatDate = (date) => {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');

  return `${year}-${month}-${day}`;
};

// Add one calendar day
const addCalendarDay = (dateString) => {
  const date = new Date(`${dateString}T12:00:00`);
  date.setDate(date.getDate() + 1);
  return formatDate(date);
};

// Return working dates, including the starting date.
// Fridays are skipped and are not counted as workdays.
const getWorkingDates = (startDate, durationDays) => {
  const workingDates = [];
  let currentDate = startDate;

  while (workingDates.length < durationDays) {
    if (!isFriday(currentDate)) {
      workingDates.push(currentDate);
    }

    currentDate = addCalendarDay(currentDate);
  }

  return workingDates;
};

// Add working days and return the final working date
const addWorkingDays = (startDate, durationDays) => {
  const dates = getWorkingDates(startDate, durationDays);
  return dates[dates.length - 1];
};

// Check workshop capacity on every working day
const checkCapacity = async (
  startDate,
  durationDays,
  settings,
  excludeId = null
) => {
  if (isFriday(startDate)) {
    return {
      available: false,
      message: 'MotoGT Workshop is closed on Fridays.'
    };
  }

  const workingDates = getWorkingDates(startDate, durationDays);

  for (const currentDate of workingDates) {
    const dateFilter = {
      status: 'Approved',
      preferredStartDate: { $lte: currentDate },
      endDate: { $gte: currentDate }
    };

    if (excludeId) {
      dateFilter._id = { $ne: excludeId };
    }

    const activeAppointments = await Appointment.find(dateFilter);

    const vehicleCount = activeAppointments.length;
    const activeWorkCount = activeAppointments.length;

    if (vehicleCount >= settings.maxVehicles) {
      return {
        available: false,
        message: `${currentDate}: Workshop vehicle capacity is full.`
      };
    }

    if (activeWorkCount >= settings.dailyWorkCapacity) {
      return {
        available: false,
        message: `${currentDate}: Daily work capacity is full.`
      };
    }
  }

  return { available: true };
};

// Customer: Create appointment request
const createAppointment = async (req, res) => {
  try {
    const {
      phone,
      vehicleModel,
      service,
      problemDescription,
      preferredStartDate
    } = req.body;
    

    if (!phone || !vehicleModel || !service || !preferredStartDate) {
      return res.status(400).json({
        message: 'Please complete all required fields.'
      });
    }

    const user = await User.findById(req.user.id).select(
      'fullName email phone'
    );

    if (!user) {
      return res.status(404).json({
        message: 'Customer account not found. Please login again.'
      });
    }

    const selectedDate = new Date(`${preferredStartDate}T12:00:00`);

    if (
      Number.isNaN(selectedDate.getTime()) ||
      formatDate(selectedDate) !== preferredStartDate
    ) {
      return res.status(400).json({
        message: 'Please select a valid appointment date.'
      });
    }

    const today = formatDate(new Date());

    if (preferredStartDate < today) {
      return res.status(400).json({
        message: 'Past dates cannot be selected.'
      });
    }

    if (isFriday(preferredStartDate)) {
      return res.status(400).json({
        message: 'MotoGT Workshop is closed on Fridays. Please select another date.'
      });
    }

    const settings = await getWorkshopSettings();
    const capacity = await checkCapacity(
      preferredStartDate,
      1,
      settings
    );

    if (!capacity.available) {
      return res.status(409).json({
        message: capacity.message
      });
    }

    const appointment = await Appointment.create({
      userId: user._id,
      customerName: user.fullName,
      email: user.email,
      phone: phone.trim(),
      vehicleModel: vehicleModel.trim(),
      service,
      problemDescription: problemDescription || '',
      preferredStartDate,
      status: 'Pending'
    });

    return res.status(201).json({
      message: 'Appointment request submitted successfully.',
      appointment
    });

  } catch (error) {
    console.error('Create Appointment Error:', error);

    return res.status(500).json({
      message: 'Unable to submit appointment request.'
    });
  }
};

// Customer: Get available start dates
const getAvailableDates = async (req, res) => {
  try {
    const settings = await getWorkshopSettings();
    const today = new Date();
    const availableDates = [];

    for (let i = 0; i < 60; i++) {
      const date = new Date(today);
      date.setDate(today.getDate() + i);

      const dateString = formatDate(date);

      // Skip Fridays
      if (isFriday(dateString)) {
        continue;
      }

      const result = await checkCapacity(
        dateString,
        1,
        settings
      );

      if (result.available) {
        availableDates.push(dateString);
      }
    }

    return res.status(200).json(availableDates);

  } catch (error) {
    console.error('Get Available Dates Error:', error);

    return res.status(500).json({
      message: 'Unable to fetch available dates.'
    });
  }
};

// Admin: Approve appointment and reserve capacity
const approveAppointment = async (req, res) => {
  let lockAcquired = false;
  let settings = null;

  try {
    const { durationDays, adminNote } = req.body;

    if (![2, 3, 4].includes(Number(durationDays))) {
      return res.status(400).json({
        message: 'Duration must be 2, 3, or 4 working days.'
      });
    }

    const appointment = await Appointment.findById(req.params.id);

    if (!appointment) {
      return res.status(404).json({
        message: 'Appointment not found.'
      });
    }

    if (appointment.status !== 'Pending') {
      return res.status(400).json({
        message: 'Only pending appointments can be approved.'
      });
    }

    if (isFriday(appointment.preferredStartDate)) {
      return res.status(400).json({
        message: 'Appointments cannot start on Friday.'
      });
    }

    settings = await getWorkshopSettings();

    const lock = await WorkshopSettings.findOneAndUpdate(
      {
        _id: settings._id,
        approvalLock: 0
      },
      {
        $set: { approvalLock: 1 }
      },
      { new: true }
    );

    if (!lock) {
      return res.status(409).json({
        message: 'Another approval is in progress. Please try again.'
      });
    }

    lockAcquired = true;

    const currentAppointment = await Appointment.findOne({
      _id: appointment._id,
      status: 'Pending'
    });

    if (!currentAppointment) {
      return res.status(409).json({
        message: 'This appointment has already been processed.'
      });
    }

    const capacity = await checkCapacity(
      currentAppointment.preferredStartDate,
      Number(durationDays),
      settings
    );

    if (!capacity.available) {
      return res.status(409).json({
        message: capacity.message
      });
    }

    currentAppointment.durationDays = Number(durationDays);

    currentAppointment.endDate = addWorkingDays(
      currentAppointment.preferredStartDate,
      Number(durationDays)
    );

    currentAppointment.status = 'Approved';
    currentAppointment.adminNote = adminNote || '';

    await currentAppointment.save();

    return res.status(200).json({
      message: 'Appointment approved and workshop capacity reserved.',
      appointment: currentAppointment
    });

  } catch (error) {
    console.error('Approve Appointment Error:', error);

    return res.status(500).json({
      message: 'Unable to approve appointment.'
    });

  } finally {
    if (lockAcquired && settings) {
      await WorkshopSettings.updateOne(
        {
          _id: settings._id,
          approvalLock: 1
        },
        {
          $set: { approvalLock: 0 }
        }
      );
    }
  }
};

// Admin: Reject appointment
const rejectAppointment = async (req, res) => {
  try {
    const appointment = await Appointment.findById(req.params.id);

    if (!appointment) {
      return res.status(404).json({
        message: 'Appointment not found.'
      });
    }

    if (appointment.status !== 'Pending') {
      return res.status(400).json({
        message: 'Only pending appointments can be rejected.'
      });
    }

    appointment.status = 'Rejected';
    appointment.adminNote = req.body.adminNote || '';

    await appointment.save();

    return res.status(200).json({
      message: 'Appointment rejected.',
      appointment
    });

  } catch (error) {
    console.error('Reject Appointment Error:', error);

    return res.status(500).json({
      message: 'Unable to reject appointment.'
    });
  }
};

// Admin: Complete appointment and release capacity
const completeAppointment = async (req, res) => {

// Admin: Update workshop capacity settings
const updateWorkshopSettings = async (req, res) => {
  try {
    const { maxVehicles, dailyWorkCapacity } = req.body;

    if (
      !Number.isInteger(Number(maxVehicles)) ||
      !Number.isInteger(Number(dailyWorkCapacity)) ||
      Number(maxVehicles) < 1 ||
      Number(dailyWorkCapacity) < 1 ||
      Number(dailyWorkCapacity) > Number(maxVehicles)
    ) {
      return res.status(400).json({
        message: 'Please provide valid workshop capacity values.'
      });
    }

    const settings = await getWorkshopSettings();

    const updatedSettings = await WorkshopSettings.findByIdAndUpdate(
      settings._id,
      {
        $set: {
          maxVehicles: Number(maxVehicles),
          dailyWorkCapacity: Number(dailyWorkCapacity)
        }
      },
      {
        new: true,
        runValidators: true
      }
    );

    return res.status(200).json({
      message: 'Workshop settings updated successfully.',
      settings: updatedSettings
    });

  } catch (error) {
    console.error('Update Workshop Settings Error:', error);

    return res.status(500).json({
      message: 'Unable to update workshop settings.'
    });
  }
};

const jwt = require("jsonwebtoken");

const protect = (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return res.status(401).json({
        message: "Not authorized. Please login.",
      });
    }

    const token = authHeader.split(" ")[1];

    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    req.user = decoded;

    next();
  } catch (error) {
    return res.status(401).json({
      message: "Invalid or expired token.",
    });
  }
};

// Recalculate membership from successfully completed orders
const syncMembership = async (userId) => {
  if (!userId) return;

  const result = await Order.aggregate([
    {
      $match: {
        userId,
        status: { $in: ['Completed', 'Delivered'] }
      }
    },
    {
      $group: {
        _id: '$userId',
        totalSpent: { $sum: '$totalAmount' }
      }
    }
  ]);

  const totalSpent = Number(result[0]?.totalSpent || 0);
  const user = await User.findById(userId);

  if (!user) return;

  if (!user.membership) {
    user.membership = {
      isActive: true,
      level: 'standard',
      discountPercentage: 5,
      totalSpent: 0
    };
  }

  user.membership.totalSpent = totalSpent;

  if (totalSpent >= 50000) {
    user.membership.level = 'vip';
    user.membership.discountPercentage = 20;
  } else if (totalSpent >= 25000) {
    user.membership.level = 'gold';
    user.membership.discountPercentage = 15;
  } else if (totalSpent >= 10000) {
    user.membership.level = 'silver';
    user.membership.discountPercentage = 10;
  } else {
    user.membership.level = 'standard';
    user.membership.discountPercentage = 5;
  }

  await user.save();
};

// Customer: Create order
const createOrder = async (req, res) => {
  const reservedProducts = [];

  try {
    const { products, address, paymentMethod = 'COD' } = req.body;

    const allowedPaymentMethods = [
      'COD',
      'bKash',
      'Nagad',
      'Card'
    ];

    if (!allowedPaymentMethods.includes(paymentMethod)) {
      return res.status(400).json({
        message: 'Invalid payment method'
      });
    }

    if (!address?.trim() || !Array.isArray(products) || !products.length) {
      return res.status(400).json({
        message: 'Delivery address and products are required'
      });
    }

    const user = await User.findById(req.user.id);

    if (!user) {
      return res.status(404).json({
        message: 'Customer not found'
      });
    }

    const quantities = new Map();

    for (const item of products) {
      const id = String(item.productId || '');
      const quantity = Number(item.quantity);

      if (!id || !Number.isInteger(quantity) || quantity < 1) {
        return res.status(400).json({
          message: 'Invalid product or quantity'
        });
      }

      quantities.set(id, (quantities.get(id) || 0) + quantity);
    }

    let subtotal = 0;
    const orderProducts = [];

    for (const [productId, quantity] of quantities) {
      const product = await Product.findOneAndUpdate(
        {
          _id: productId,
          stock: { $gte: quantity }
        },
        {
          $inc: { stock: -quantity }
        },
        { new: true }
      );

      if (!product) {
        throw new Error(
          `Product unavailable or insufficient stock: ${productId}`
        );
      }

      reservedProducts.push({
        productId: product._id,
        quantity
      });

      subtotal += product.price * quantity;

      orderProducts.push({
        productId: product._id,
        quantity
      });
    }

    const discountPercentage = user.membership?.isActive
      ? (user.membership.discountPercentage || 5)
      : 0;

    if (
      ![5, 10, 15, 20].includes(discountPercentage) &&
      discountPercentage !== 0
    ) {
      throw new Error('Invalid membership discount');
    }

    const discountAmount = Number(
      (subtotal * discountPercentage / 100).toFixed(2)
    );

    const totalAmount = Number(
      (subtotal - discountAmount).toFixed(2)
    );

    const order = await Order.create({
      userId: user._id,
      customerName: user.fullName,
      email: user.email,
      phone: user.phone,
      address: address.trim(),
      products: orderProducts,
      subtotal,
      discountPercentage,
      discountAmount,
      totalAmount,
      paymentMethod,
      paymentStatus: 'Pending',
      status: 'Pending'
    });

    res.status(201).json({
      message: 'Order created successfully',
      order
    });

  } catch (error) {
    for (const item of reservedProducts) {
      await Product.updateOne(
        { _id: item.productId },
        { $inc: { stock: item.quantity } }
      ).catch(() => {});
    }

    res.status(400).json({ message: error.message });
  }
};

            cx="13.5"

            stroke="currentColor"
            strokeWidth="2.6"

            cx="36"

                      className={

                          ? "product-stock available"
                          : "product-stock unavailable"

                      className="accessories-cart-item"
                      key={product._id}
                    >
                      {product.image && (
                        <img
                          src={product.image}
                          alt={product.name}
                        />
                      )}

                      <div className="accessories-cart-item-info">
                        <strong>{product.name}</strong>

                        <span>
                          ৳
                          {Number(
                            product.price
                          ).toLocaleString("en-BD")}
                        </span>

                        {product.description && (
                          <p className="accessories-cart-description">
                            {product.description}
                          </p>
                        )}

                        <div className="accessories-quantity-control">
                          <button
                            type="button"
                            onClick={() =>
                              changeQuantity(product._id, -1)
                            }
                          >
                            −
                          </button>

                          <span>{quantity}</span>

                          <button
                            type="button"
                            onClick={() =>
                              changeQuantity(product._id, 1)
                            }
                          >
                            +
                          </button>

                          <button
                            type="button"
                            className="accessories-remove-button"
                            onClick={() =>
                              removeFromCart(product._id)
                            }
                          >
                            Remove
                          </button>
                        </div>
                      </div>
                    </div>
                  ))}
                </div>

                {/* Cart subtotal */}
                <div className="accessories-cart-total">
                  <span>Subtotal</span>

                  <strong>
                    ৳
                    {subtotal.toLocaleString("en-BD", {
                      minimumFractionDigits: 2,
                      maximumFractionDigits: 2,
                    })}
                  </strong>
                </div>

                {/* Membership discount */}
                <div className="accessories-cart-total">
                  <span>
                    Membership Discount
                    {membershipLoading
                      ? " (Loading...)"
                      : ` (${discountPercentage}%)`}
                  </span>

                  <strong>
                    - ৳
                    {discountAmount.toLocaleString("en-BD", {
                      minimumFractionDigits: 2,
                      maximumFractionDigits: 2,
                    })}
                  </strong>
                </div>

                {/* Final total */}
                <div className="accessories-cart-total">
                  <strong>Final Total</strong>

                  <strong>
                    ৳
                    {finalTotal.toLocaleString("en-BD", {
                      minimumFractionDigits: 2,
                      maximumFractionDigits: 2,
                    })}
                  </strong>
                </div>

                {/* Checkout */}
                <form
                  className="accessories-checkout-form"
                  onSubmit={placeOrder}
                >
                  <label htmlFor="delivery-address">
                    Delivery Address
                  </label>

                  <textarea
                    id="delivery-address"
                    value={address}
                    onChange={(event) =>
                      setAddress(event.target.value)
                    }
                    placeholder="Enter your full delivery address"
                    required
                    rows={3}
                  />

                  <label className="accessories-payment-label">
                    Payment Method
                  </label>

                  <div className="accessories-payment-options">
                    {paymentOptions.map((option) => (
                      <button
                        type="button"
                        key={option.value}
                        className={`accessories-payment-option ${
                          paymentMethod === option.value
                            ? "selected"
                            : ""
                        }`}
                        onClick={() =>
                          setPaymentMethod(option.value)
                        }
                        disabled={placingOrder}
                        aria-pressed={
                          paymentMethod === option.value
                        }
                      >
                        <span
                          className={`payment-option-logo ${option.type}-logo`}
                        >
                          {renderPaymentMark(option)}
                        </span>

                        <span className="payment-option-copy">
                          <strong>{option.name}</strong>
                          <small>{option.detail}</small>
                        </span>

                        <span className="payment-option-check">
                          {paymentMethod === option.value ? "✓" : ""}
                        </span>
                      </button>
                    ))}
                  </div>

                  {orderError && (
                    <p className="accessories-order-error">
                      {orderError}
                    </p>
                  )}

                  <button
                    type="submit"
                    disabled={
                      placingOrder ||
                      cart.length === 0 ||
                      Boolean(pendingPaymentOrderId)
                    }
                  >
                    {placingOrder
                      ? "Placing Order..."
                      : "Place Order"}
                  </button>
                </form>
              </>
            )}

            {/* Demo payment confirmation */}
            {pendingPaymentOrderId && (
              <div className="accessories-demo-payment">
                <h3>Demo Payment Confirmation</h3>

                <p>
                  Order ID: <strong>{pendingPaymentOrderId}</strong>
                </p>

                <p>
                  Payment Method:{" "}
                  <strong>{pendingPaymentMethod}</strong>
                </p>

                <p>
                  Payment Status: <strong>Pending</strong>
                </p>

                <p>
                  This is a simulated payment. Press the button
                  below to confirm a successful demo transaction.
                </p>

                <button
                  type="button"
                  onClick={confirmDemoPayment}
                  disabled={confirmingPayment}
                >
                  {confirmingPayment
                    ? "Confirming Payment..."
                    : "Confirm Demo Payment"}
                </button>
              </div>
            )}

            {orderMessage && (
              <p className="accessories-order-success">
                {orderMessage}
              </p>
            )}
          </aside>
        </div>
      )}
    </section>
  );
}

const User = require("../models/User");

// Get logged-in customer's profile and membership
router.get("/profile", protect, async (req, res) => {
  try {
    const user = await User.findById(req.user.id).select("-password");

    if (!user) {
      return res.status(404).json({
        message: "Customer not found",
      });
    }

    res.status(200).json({
      message: "Profile loaded successfully",
      user,
    });
  } catch (error) {
    res.status(500).json({
      message: error.message,
    });
  }
});

import { useState } from "react";
import logo from "../assets/MotoGT-logo-clean.png";
import { apiFetch } from "../utils/api";
import "../styles/Auth.css";

function Login() {
  const [formData, setFormData] = useState({
    email: "",
    password: "",
  });

  const [message, setMessage] = useState("");
  const [loading, setLoading] = useState(false);

  const handleChange = (e) => {
    setFormData({
      ...formData,
      [e.target.name]: e.target.value,
    });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setMessage("");
    setLoading(true);

    try {
      const response = await apiFetch("/auth/login", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify(formData),
      });

      const data = await response.json().catch(() => ({}));

      if (!response.ok) {
        throw new Error(data.message || "Login failed.");
      }

      localStorage.setItem("motogtToken", data.token);
      localStorage.setItem("motogtUser", JSON.stringify(data.user));

      setMessage("Login successful!");
      window.location.href = "/";
    } catch (error) {
      setMessage(error.message || "Something went wrong.");
    } finally {
      setLoading(false);
    }
  };

  return (
    <section className="auth-section">
      <div className="auth-card">
        <img src={logo} alt="MotoGT Logo" className="auth-logo" />

        <p className="auth-label">WELCOME BACK</p>

        <h1>Login to Your <span>Account.</span></h1>

        <p className="auth-description">
          Sign in to continue your journey with us.
        </p>

        <form className="auth-form" onSubmit={handleSubmit}>
          <label>Email Address</label>
          <input
            type="email"
            name="email"
            placeholder="Enter your email"
            value={formData.email}
            onChange={handleChange}
            required
          />

          <label>Password</label>
          <input
            type="password"
            name="password"
            placeholder="Enter your password"
            value={formData.password}
            onChange={handleChange}
            required
          />

          {message && <p>{message}</p>}

          <button type="submit" disabled={loading}>
            {loading ? "Logging in..." : "Login"}
          </button>
        </form>

        <p className="auth-switch">
          Don't have an account?{" "}
          <a href="/register">Create Account</a>
        </p>

        <p className="guest-option">
          Or <a href="/">Continue as Guest</a>
        </p>
      </div>
    </section>
  );
}

export default Login;
